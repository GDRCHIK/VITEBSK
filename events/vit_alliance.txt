	id = vit_alliance.4
	title = vit_alliance.4.t
	desc = vit_alliance.4.d
	picture = GFX_news_event_generic_sign_treaty1 # можешь поменять картинку
	
	is_triggered_only = yes # Важно! Событие сработает только когда его вызовут из фокуса

	option = {
		name = vit_alliance.4.a # Принять
		ai_chance = { base = 50 } # Шанс 50%

			country_event = { id = vit_alliance.5 } # Отправляем ответ Витебску

		}
	}
	option = {
		name = vit_alliance.4.b # Отказать
		ai_chance = { base = 50 } # Шанс 50%
		VIT = { 
			country_event = { id = vit_alliance.6 } # Отправляем ответ Витебску об отказе
		}
	}
}
